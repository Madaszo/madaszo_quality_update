merge_culture_group = {
	ability = adm
	allow_multiple = no
	years = 10
	potential = {
		culture = { 
			has_any_culture_group = yes
			is_merged_culture_group = no
			any_culture_group = {
				is_already_merged = no
			}
		}
	}
	allow = {
		is_dominant_country_of = culture
		#country_rank = country_rank:rank_empire
	}

	select_trigger = {
		looking_for_a = culture_group
		target_flag = target
		name = "merge_culture_group_select_culture_group"
		show_why_not_enabled = yes
		none_available_msg_key = "merge_culture_group_no_culture_group"
		column = {
			data = name
		}
		column = {
			data = expected_assimilation
		}

		interaction_source_list = {
			scope:actor.culture = {
				every_culture_group = {
					add_to_list = source
				}
			}
		}
		visible = {

		}
		enabled = {
			# The action is available only to a country with the largest
			# population belonging to the selected culture group.
			NOT = {
				any_country = {
					limit = {
						this != scope:actor
					}
					culture_group_population_in_country = {
						culture_group = root
						value > {
							value = "scope:actor.culture_group_population_in_country(root)"
						}
					}
				}
			}
			is_already_merged = no
		}
	}
	country_modifier = {
		monthly_towards_centralization = 0.1
	}
	on_fully_activated = {
		scope:actor = {
			# Grant the bonus advances of every culture in the merged group.
			scope:target = {
				every_culture = {
					scope:actor = {
						grant_culture_bonus_advances = prev
					}
				}
			}
			merge_culture_group = scope:target
		}
	}
}
